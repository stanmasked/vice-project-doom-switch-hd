cmake_minimum_required(VERSION 3.16)
project(vpd_native_chr_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vpd_chr STATIC nes_chr.cpp)
target_include_directories(vpd_chr PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(vpd_chr PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vpd_chr_cli chr_cli.cpp)
target_link_libraries(vpd_chr_cli PRIVATE vpd_chr)
add_executable(vpd_chr_test test_core.cpp)
target_link_libraries(vpd_chr_test PRIVATE vpd_chr)
enable_testing()
add_test(NAME chr_decode_unit COMMAND vpd_chr_test)
